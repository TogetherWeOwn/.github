name: PR lint

# TogetherWeOwn GitHub standard, one-click template for new repos.
# Always an error: Conventional Commits title, and a commit on the default branch that is not one.
# PR_STANDARDS_MODE ("warn" or "error") sets the level for the rest: an empty body and the
# PR template sections (Thinking Path, What Changed, Verification, Risks, Model Used, and
# Linked Issues or Issue Description for feat/fix/perf/refactor/security). Land it in "warn",
# clear the open PRs, then flip to "error".
# Private repos: set REQUIRE_CARD_REF to "true" so every PR links its tracking card.
# Set INTERNAL_ID_PREFIXES to a comma- or pipe-separated list (e.g. "ABC,XYZ") to flag
# org-specific ticket ids; empty means a generic ticket-like pattern.
